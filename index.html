const SHEET_NAME = 'memos'; function doGet(e) { try { const faqId = e && e.parameter ? e.parameter.faq_id : ''; const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['id', 'faq_id', 'created_at', 'user_name', 'memo_text']); } const data = sheet.getDataRange().getValues(); const memos = []; for (let i = 1; i < data.length; i++) { if (String(data[i][1]) === String(faqId)) { memos.push({ id: data[i][0], faq_id: data[i][1], created_at: data[i][2] ? Utilities.formatDate(new Date(data[i][2]), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss') : '', user_name: data[i][3], memo_text: data[i][4] }); } } return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: memos })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { const contents = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['id', 'faq_id', 'created_at', 'user_name', 'memo_text']); } // 削除処理（action: 'delete' の場合） if (contents.action === 'delete') { const targetId = contents.id; const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(targetId)) { sheet.deleteRow(i + 1); return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } } throw new Error('該当するメモが見つかりませんでした'); } // 投稿処理 const now = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss'); const memoId = 'MEMO_' + new Date().getTime() + '_' + Math.floor(Math.random() * 1000); const faqId = contents.faq_id; const userName = contents.user_name || '匿名'; const memoText = contents.memo_text; if (!faqId || !memoText) { throw new Error('必須項目が不足しています'); } sheet.appendRow([memoId, faqId, now, userName, memoText]); return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
