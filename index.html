if (!response.ok) { throw new Error(`HTTP error! status: ${response.status}`); } const memos = await response.json(); // 画面のメモ表示エリアを更新する関数（ご自身の描画処理関数名に合わせてください） renderMemosUI(memos); } catch (error) { console.error("メモ取得エラー:", error); // 「メモの取得に失敗しました」と出る原因はここです showToast("メモの取得に失敗しました"); } } // 2. 【保存】iframe＋非表示フォームを使った書き込み処理（CORS回避） function saveMemo(textValue) { if (!textValue) return; let iframe = document.getElementById('hidden-iframe'); if (!iframe) { iframe = document.createElement('iframe'); iframe.id = 'hidden-iframe'; iframe.name = 'hidden-iframe'; iframe.style.display = 'none'; document.body.appendChild(iframe); } const form = document.createElement('form'); form.action = MEMO_GAS_URL; form.method = 'POST'; form.target = 'hidden-iframe'; const actionInput = document.createElement('input'); actionInput.type = 'hidden'; actionInput.name = 'action'; actionInput.value = 'add'; form.appendChild(actionInput); const textInput = document.createElement('input'); textInput.type = 'hidden'; textInput.name = 'text'; textInput.value = textValue; form.appendChild(textInput); document.body.appendChild(form); form.submit(); document.body.removeChild(form); // 送信後、少し待ってからメモ一覧を再読み込み setTimeout(() => { loadMemos(); }, 1200); } // 3. 【削除】iframe＋非表示フォームを使った削除処理（CORS回避） function deleteMemo(memoId) { let iframe = document.getElementById('hidden-iframe'); if (!iframe) { iframe = document.createElement('iframe'); iframe.id = 'hidden-iframe'; iframe.name = 'hidden-iframe'; iframe.style.display = 'none'; document.body.appendChild(iframe); } const form = document.createElement('form'); form.action = MEMO_GAS_URL; form.method = 'POST'; form.target = 'hidden-iframe'; const actionInput = document.createElement('input'); actionInput.type = 'hidden'; actionInput.name = 'action'; actionInput.value = 'delete';
